My Java Project

‎Stock Level Classifier — Store the available quantities of 15 products in an int[] ‎array, accepting nonnegative integers. Classify each product as Low Stock (below 10), ‎Moderate Stock (10–49), or High Stock (50 or more). Count products in each category. ‎Display each product number, quantity, and classification.

import java.util.Scanner; public class Main { public static void main(String[] args) { Scanner input = new Scanner(System.in); int[] stock = new int[15]; int low = 0, moderate = 0, high = 0; for (int i = 0; i < 15; i++) { System.out.print("Enter quantity of product " + (i + 1) + ": "); stock[i] = input.nextInt(); while (stock[i] < 0) { System.out.print("Enter a nonnegative number: "); stock[i] = input.nextInt(); } } System.out.println("\n===== STOCK CLASSIFICATION ====="); for (int i = 0; i < 15; i++) { System.out.print("Product " + (i + 1)); System.out.print(" - Quantity: " + stock[i]); if (stock[i] < 10) { System.out.println(" - Low Stock"); low++; } else if (stock[i] < 50) { System.out.println(" - Moderate Stock"); moderate++; } else { System.out.println(" - High Stock"); high++; } } System.out.println("\n===== TOTAL PRODUCTS ====="); System.out.println("Low Stock: " + low); System.out.println("Moderate Stock: " + moderate); System.out.println("High Stock: " + high); } }
